#   COMMANDS TO RUN WHEN SETTING UP A NEW VM:
### (printed by ChatGPT for me, but essentially the commands I ended up using all the time!)

# Update
apt update && apt upgrade -y

# Install Docker
apt install -y docker.io
systemctl enable docker
systemctl start docker

# Add group:
sudo getent group honeypotlogs || sudo groupadd -g 20001 honeypotlogs

# Change to root user (using `sudo -i` or `sudo su -`) and run the following:
sudo visudo
# Now add:
<username> ALL=(ALL) NOPASSWD: ALL
## REMEMBER TO CHANGE <username> TO THE USERNAME OF THE MAIN USER OF THE VM (e.g., "ubuntu" or "hpu")


# Add user
adduser --disabled-password --gecos "" honeypotlogs
usermod -aG docker honeypotlogs
------- OR -------
sudo id -u honeypotlogs >/dev/null 2>&1 || sudo useradd -u 20001 -g 20001 -M -d /home/honeypotlogs -s /usr/sbin/nologin honeypotlogs

# Create log directory
mkdir -p /home/<username>/honey-llm-logs
sudo chown -R honeypotlogs:honeypotlogs /home/<username>/honey-llm-logs

# Set ownerships and `chmod` commands (REMEMBER TO CHANGE USERNAME TO MATCH THE VM'S USER!!!):
sudo chown <username>:<username> ~/honey-llm-logs/
chmod 775 ~/honey-llm-logs/
sudo chown honeypotlogs:honeypotlogs ~/honey-llm-logs/honeypot-logs/
sudo chown honeypotlogs:honeypotlogs ~/honey-llm-logs/tcpdump-logs/
sudo chmod 770 ~/honey-llm-logs/honeypot-logs/
sudo chmod 2770 ~/honey-llm-logs/honeypot-logs/
sudo chmod 770 ~/honey-llm-logs/honeypot-logs/in-out.log

# Set specific permissions (REMEMBER TO CHANGE USERNAME TO MATCH THE VM'S USER!!!):
sudo setfacl -m u:hpu:rwx,u:honeypotlogs:rwx,m:rwx /home/hpu/honey-llm-logs/
sudo setfacl -d -m u:hpu:rwx,u:honeypotlogs:rwx,m:rwx /home/hpu/honey-llm-logs/
sudo setfacl -m u:hpu:rwx,u:honeypotlogs:rwx,m:rwx /home/hpu/honey-llm-logs/honeypot-logs/
sudo setfacl -d -m u:hpu:rwx,u:honeypotlogs:rwx,m:rwx /home/hpu/honey-llm-logs/honeypot-logs/
sudo setfacl -m u:hpu:rwx,u:honeypotlogs:rwx,m:rwx /home/hpu/honey-llm-logs/tcpdump-logs/
sudo setfacl -d -m u:hpu:rwx,u:honeypotlogs:rwx,m:rwx /home/hpu/honey-llm-logs/tcpdump-logs/
